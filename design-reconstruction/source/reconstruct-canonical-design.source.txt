  for (const [method, operation] of Object.entries(pathItem)) {
    if (!operation?.operationId) continue;
    openApiOperationMetadata.set(operation.operationId, {
      method: method.toUpperCase(),
      path: operationPath,
      owner: operation["x-module-owner"],
      contractStatus: operation["x-contract-status"],
      transactionBoundary: operation["x-transaction-boundary"],
      idempotency: operation["x-idempotency-policy"],

      audit: operation["x-audit-requirement"],
      outbox: operation["x-event-outbox-expectation"],
      credentials: operation["x-credentials-policy"],
    });
  }
}

const walk = (directory, exclusions = new Set()) => {
  const result = [];
  const visit = (current) => {
    for (const name of fs.readdirSync(current)) {
      if (exclusions.has(name)) continue;
      const full = path.join(current, name);
      const stat = fs.statSync(full);
      if (stat.isDirectory()) visit(full);
      else result.push(rel(path.relative(root, full)));
    }
  };
  visit(path.join(root, directory));
  return result.sort();
};

const exportedDeclarations = (files, selector) => {
  const names = new Set();
  for (const file of files.filter((value) => /\.tsx?$/.test(value) && selector(value))) {
    const source = read(file);
    const sf = ts.createSourceFile(file, source, ts.ScriptTarget.Latest, true, file.endsWith(".tsx") ? ts.ScriptKind.TSX : ts.ScriptKind.TS);
    for (const node of sf.statements) {
      if (!node.modifiers?.some((modifier) => modifier.kind === ts.SyntaxKind.ExportKeyword)) continue;
      if ((ts.isFunctionDeclaration(node) || ts.isInterfaceDeclaration(node) || ts.isTypeAliasDeclaration(node) || ts.isEnumDeclaration(node) || ts.isClassDeclaration(node)) && node.name) names.add(node.name.text);
      if (ts.isVariableStatement(node)) for (const declaration of node.declarationList.declarations) if (ts.isIdentifier(declaration.name)) names.add(declaration.name.text);
    }
  }
  return [...names].sort();
};

"repository", 1, [file.startsWith("docs/") ? "EV-INVENTORY" : "EV-MODULES"]];
  if (file === "package-lock.json") return ["PROVENANCE_ONLY", "Dependency lock provenance", "repository", 0, ["EV-INVENTORY"]];
  return ["READ_SUMMARY", "Repository configuration/deployment provenance reviewed by census", "repository", 1, ["EV-INVENTORY"]];
};

const buildCoverage = () => {
  const files = walk(".", exclusions);
  const entries = files.map((file) => {
    const stat = fs.statSync(path.join(root, file));
    const [classification, reason, owner, dependencyRadius, evidenceRefs] = classify(file);
    const content = fs.readFileSync(path.join(root, file));
    return { path: file, classification, owner, reason, analyzed: classification !== "EXCLUDED_WITH_REASON", evidenceRefs, dependencyRadius, bytes: stat.size, lines: /\.(md|json|js|mjs|ts|tsx|css|html|yml|yaml|txt)$/.test(file) ? content.toString("utf8").split(/\r?\n/).length : null };
  });
  const byClassification = Object.fromEntries(Object.entries(Object.groupBy(entries, (entry) => entry.classification)).map(([key, values]) => [key, values.length]));
  const sourceEntries = entries.filter((entry) => !entry.path.startsWith("canonical-design/") && !entry.path.startsWith("design-reconstruction/"));
  const fingerprint = sha256(sourceEntries.map((entry) => `${entry.path}:${entry.bytes}`).join("\n"));
  writeJson("design-reconstruction/source/source-coverage-ledger.json", { schemaVersion: 1, status: "COMPLETE_CLASSIFICATION", generatedAt: new Date().toISOString(), exclusions: [...exclusions], sourceSnapshotFiles: sourceEntries.length, repositoryFilesIncludingOutputs: entries.length, unclassifiedFiles: 0, byClassification, sourceFingerprint: fingerprint, files: entries });
  writeJson("design-reconstruction/source/repository-manifest.json", { schemaVersion: 1, sourcePackage: "unicorecrm-web@0.23.20-contract.0", actualSourceFiles: sourceEntries.length, storedInventoryFiles: inventory.summary.repositoryFiles, storedInventoryFingerprint: inventory.inventoryFingerprint, actualSourceFingerprint: fingerprint, storedInventoryDrift: sourceEntries.length !== inventory.summary.repositoryFiles, protectedInventory: { modules: inventory.summary.modules, routes: inventory.summary.routes, loadableRouteModules: inventory.summary.loadableRouteModules, capabilities: inventory.summary.capabilities, workspaceFlags: inventory.summary.workspaceFlags, workflows: inventory.summary.workflows }, openapi: { version: openapi.info.version, sha256: openapiHash, operations: operationCatalog.operations.length, ready: readyOperations.length, blocked: blockedOperations.length, schemas: Object.keys(openapi.components?.schemas ?? {}).length }, commands: commandStatusCounts, queries: queryClassCounts });
  return { entries, sourceEntries, fingerprint, byClassification };
};
